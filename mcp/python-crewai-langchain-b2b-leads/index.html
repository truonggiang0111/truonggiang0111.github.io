CrewAI BaseTool + Agent + Crew (Python 3.10+) 
from crewai import Agent, Crew, Task
from crewai.tools import BaseTool
import os, requests

class ShopifyChurnSpyTool(BaseTool):
    name: str = "shopify_app_churn_spy"
    description: str = "Mine 1-2 star Shopify app churn reviews, complaining merchant domains, and dev emails."

    def _run(self, query: str) -> list[dict]:
        resp = requests.post(
            "https://api.apify.com/v2/acts/ttyan0111~shopify-app-store-churn-spy/run-sync-get-dataset-items",
            params={"token": os.environ["APIFY_TOKEN"], "fields": "name,developerEmail,mentionedDomains,painSignalSummary,agentSummary"},
            json={"queries": [query], "starRatings": ["1", "2"], "maxAppsPerQuery": 3, "maxReviewsPerApp": 15},
            timeout=90,
        )
        return resp.json()

analyst = Agent(
    role="B2B Churn Intelligence Analyst",
    goal="Identify merchants leaving 1-star app reviews for targeted outreach.",
    backstory="Extracts verified product-led outbound triggers from public app reviews.",
    tools=[ShopifyChurnSpyTool()],
)
task = Task(description="Audit 1-star churn around 'klaviyo-email-marketing'.", expected_output="JSON lead brief.", agent=analyst)
print(Crew(agents=[analyst], tasks=[task]).kickoff())

LangChain StructuredTool + ChatOpenAI.bind_tools() 
from langchain_core.tools import StructuredTool
from langchain_openai import ChatOpenAI
from uk_tenders_procurement_agent import search_uk_gov_tenders_zero_key

uk_tenders_tool = StructuredTool.from_function(
    func=search_uk_gov_tenders_zero_key,
    name="search_uk_gov_tenders",
    description="Search UK Contracts Finder notices returning buyer name, .gov.uk email, GBP value & deadline (zero keys).",
)

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0).bind_tools([uk_tenders_tool])
ai_msg = llm.invoke("Find 3 open UK public sector cloud tenders with buyer contact emails.")
for call in ai_msg.tool_calls:
    print(search_uk_gov_tenders_zero_key(**call["args"]))

OpenAI Tool-Calling Loop + Zero-Key Google News Decoder 
import json
from openai import OpenAI
from google_news_rag_agent import OPENAI_TOOLS, dispatch_tool

client = OpenAI()
messages = [
    {"role": "system", "content": "Resolve opaque Google News links before summarizing."},
    {"role": "user", "content": "Decode https://news.google.com/rss/articles/... and cite the real publisher."},
]
resp = client.chat.completions.create(model="gpt-4o-mini", messages=messages, tools=OPENAI_TOOLS)
for call in resp.choices[0].message.tool_calls or []:
    decoded = dispatch_tool(call.function.name, json.loads(call.function.arguments))
    print(decoded["decoded_url"], "via", decoded["method"])
