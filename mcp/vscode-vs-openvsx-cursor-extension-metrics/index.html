Option B — Python Script: Cross-Check VS Code Marketplace vs Open-VSX Installs

benchmark_vscode_openvsx.py 
import requests

VSCODE_URL = "https://marketplace.visualstudio.com/_apis/public/gallery/extensionquery"
ext_id = "anthropic.claude-code"
pub, name = ext_id.split(".", 1)

vsc_resp = requests.post(
    VSCODE_URL,
    headers={"Accept": "application/json;api-version=7.1-preview.1", "Content-Type": "application/json"},
    json={
        "filters": [{"criteria": [{"filterType": 7, "value": ext_id}], "pageNumber": 1, "pageSize": 1}],
        "assetTypes": [],
        "flags": 918,
    },
    timeout=20,
).json()

ext = vsc_resp["results"][0]["extensions"][0]
vsc_installs = int(next((s["value"] for s in ext["statistics"] if s["statisticName"] == "install"), 0))
ovx_resp = requests.get(f"https://open-vsx.org/api/{pub}/{name}", timeout=20).json()
ovx_downloads = int(ovx_resp.get("downloadCount", 0))

print({
    "extension": ext_id,
    "verifiedDomain": ext["publisher"].get("domain"),
    "vscodeInstalls": vsc_installs,
    "openVsxDownloads": ovx_downloads,
    "totalCombinedInstalls": vsc_installs + ovx_downloads,
})
